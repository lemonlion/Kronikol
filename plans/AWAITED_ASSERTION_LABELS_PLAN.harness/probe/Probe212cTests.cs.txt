    public void Probe()
    {
        var watch = Stopwatch.StartNew();
        var results = new StringBuilder();
        results.AppendLine("# Probe212c: plain Release against Release+debug-plus, each built by `dotnet build`, SDK " + SdkVersion + ", " + Tfm);
        results.AppendLine(Versions());
        results.AppendLine($"Fixtures are built under {OutputDir}, one project directory per build (a shared obj/ lets the second build skip CoreCompile).");

        // 0. The compiler's switches, read from each build's own output.
        results.AppendLine();
        results.AppendLine("## 0. The csc command line of the same fixture (the first probe's S2) built both ways, read from `dotnet build -v:d`");
        var confirm = new StringBuilder();
        var s2 = FirstProbeShapes.Single(s => s.Id == "S2");
        foreach (var build in new[] { Plain, DebugPlus })
        {
            var name = $"Probe212c_confirm_{build.Tag}";
            try
            {
                var (_, output, command) = BuildFixture(name, Source(s2, "first"), build, tunit: false, verbosity: "d");
                results.AppendLine();
                results.AppendLine($"### {build.Label}");
                results.AppendLine("dotnet " + command);
                var csc = output.ReplaceLineEndings("\n").Split('\n').Where(IsCscCommandLine).ToList();
                results.AppendLine($"csc command line ({csc.Count} found):");
                foreach (var line in csc)
                    results.AppendLine("  " + line.Trim());
                var named = output.ReplaceLineEndings("\n").Split('\n')
                    .Where(l => Regex.IsMatch(l, @"DebugSymbols|EmitDebugInformation")).Select(l => l.Trim()).Distinct().ToList();
                results.AppendLine($"lines of the -v:d output naming DebugSymbols or EmitDebugInformation ({named.Count}):");
                foreach (var line in named)
                    results.AppendLine("  " + Cut(line, 400));
                var switches = Switches(output);
                results.AppendLine("switches: " + switches);
                confirm.AppendLine($"- {build.Label}: csc {switches} (`{name}`, -v:d)");
            }
            catch (Exception ex)
            {
                results.AppendLine($"!! {build.Label}: {ex.Message.ReplaceLineEndings(" | ")}");
                confirm.AppendLine($"- {build.Label}: FAILED: {Cut(ex.Message.ReplaceLineEndings(" "), 300)}");
            }
        }

        // 1. Each shape, built both ways.
        var shapes = Wanted.Select(id =>
        {
            var first = FirstProbeShapes.FirstOrDefault(s => s.Id == id);
            if (first != null)
                return (Shape: first, Template: first.FullSource != null ? "full" : "first");
            return (Shape: BShapes.Single(s => s.Id == id), Template: "b");
        }).ToList();
        // S24 (a suite's own [ShouldlyMethods] async assertion) is built in Debug as well.
        shapes.AddRange(S24Shapes.Select(s => (Shape: s, Template: "s24")));
        var measured = new List<(Shape Shape, string Template, Measured Plain, Measured DebugPlus, Measured? Debug)>();
        foreach (var (shape, template) in shapes)

            results.AppendLine();
            results.AppendLine($"## {shape.Id} ({template switch { "b" => "Probe212b's template", "full" => "Probe212's verbatim fixture", "s24" => "Probe212b's template with a [ShouldlyMethods] type of the fixture's own", _ => "Probe212's template" }}) {shape.Title}");
            results.AppendLine("source:");
            results.Append(Numbered(Source(shape, template)));
            var debug = template == "s24" ? Measure(shape, template, DebugBuild, results) : null;

            measured.Add((shape, template, plain, debugPlus, debug));
        }
        File.WriteAllText(Path.Combine(Harness, "results-212c-debugplus.txt"), results.ToString());

        $"Kronikol {Informational(typeof(Track).Assembly)}, Kronikol.AssertionTracking {Informational(typeof(AssertionWeaver).Assembly)}, " +
        $"Shouldly {Informational(typeof(Shouldly.Should).Assembly)}, FluentAssertions {Informational(typeof(FluentAssertions.AssertionExtensions).Assembly)}, " +
        $"TUnit.Assertions {Informational(typeof(TUnit.Assertions.Assert).Assembly)}, runtime {Environment.Version}; " +
        $"SDK {SdkVersion}'s compiler Roslyn {SdkRoslyn()}";

    private static Measured Measure(Shape shape, string template, Build build, StringBuilder results)
    {
        var m = new Measured();
        var name = $"Probe212c_{shape.Id}_{build.Tag}";
        var source = Source(shape, template);
        var entries = shape.Entries ?? ["Run"];
        m.Entries.AddRange(entries);
        results.AppendLine();
        results.AppendLine($"### {shape.Id} [{build.Label}]");

        string path, output, command;
        try
        {
            (path, output, command) = BuildFixture(name, source, build, shape.TUnit, verbosity: "n");
        }
        catch (Exception ex) when (ex is InvalidOperationException or FileNotFoundException)
        {
            m.Failed = ex.Message.ReplaceLineEndings(" | ");
            results.AppendLine("build: FAILED: " + m.Failed);
            return m;
        }
        m.Csc = Switches(output);
        results.AppendLine($"build: dotnet {command}");
        results.AppendLine($"csc switches from the build's own output: {m.Csc}");
        var lines = File.ReadAllLines(Path.ChangeExtension(path, ".cs"));

        // The unwoven copy, kept before the weave: every build is named WeaverFixture, so each run loads its assembly
        // into a load context of its own.
        var plainPath = Path.Combine(Path.GetDirectoryName(path)!, name + "_plain.dll");
        File.Copy(path, plainPath, overwrite: true);
        File.Copy(Path.ChangeExtension(path, ".pdb"), Path.ChangeExtension(plainPath, ".pdb"), overwrite: true);

        // 1. The original IL, and the weaver's own reading of it.
        var plan = new StringBuilder();
        var entryPoints = new StringBuilder();
        var awaits = new StringBuilder();
        using (var module = Read(path))
        {
            var types = module.GetTypes().Where(t => IsUnder(t, shape.TypeName)).ToList();
            var weaver = new AssertionWeaver();
            foreach (var type in types)
            {
                foreach (var method in type.Methods.Where(mt => mt.HasBody).ToList())
                {
                    EntryPoints(method, entryPoints, lines, m);
                    var statements = (List<AssertionStatement>)FindStatements.Invoke(weaver, [method, new WeaveResult()])!;
                    var excerpts = statements.ToDictionary(s => s, s => Excerpt(method, s, lines));
                    method.Body.SimplifyMacros();
                    var analysis = StackAnalysis.Analyze(method.Body);
                    foreach (var st in statements)
                    {
                        var sp = st.SequencePoint;
                        var key = $"{type.Name}::{method.Name} L{sp.StartLine}";
                        plan.AppendLine($"  {type.Name}::{method.Name} L{sp.StartLine}:{sp.StartColumn}-L{sp.EndLine}:{sp.EndColumn}  awaited={st.IsAwaited}  GetResult={(st.GetResultInstruction is { } gr ? $"IL_{gr.Offset:x4} {Operand(gr)}" : "none")}  stack analysis consistent={analysis.Consistent}");
                        plan.AppendLine($"    SourceText: \"{Esc(st.SourceText)}\"");
                        plan.AppendLine($"    statement IL_{st.FirstInstruction.Offset:x4} {st.FirstInstruction.OpCode} .. IL_{st.LastInstruction.Offset:x4} {st.LastInstruction.OpCode} {Operand(st.LastInstruction)}");
                        foreach (var cv in st.CapturedVariables)
                            plan.AppendLine($"    captured value {Describe(cv, method)}");
                        if (st.CapturedVariables.Count == 0)
                            plan.AppendLine("    captured value: none");
                        var (display, verdict, stacks, next) = PlanOf(method, st, analysis, plan);
                        m.Statements[key] = display;
                        m.Verdicts[key] = verdict;
                        m.PlanStacks[key] = stacks;
                        if (next != null)
                            m.RangeNext[key] = next;
                        if (excerpts[st] is { } excerpt)
                        {
                            plan.AppendLine("    original IL from the statement's GetAwaiter to the next visible sequence point:");
                            plan.Append(excerpt);
                        }
                    }
                    Awaits(method, analysis, lines, awaits, m);
                }
            }
        }

        // 2. The weave itself, and what it added that ILVerify rejects.
        var before = File.ReadAllBytes(path);
        var weave = new AssertionWeaver().Weave(path, Path.ChangeExtension(path, ".pdb"));
        var added = WovenIl.NewErrors(before, File.ReadAllBytes(path), path);
        m.Woven = weave.WeavedCount;
        m.Unwoven = weave.Unwoven.Select(ShortReason).OrderBy(r => r, StringComparer.Ordinal).ToList();
        m.Verify = added.Count == 0 ? "clean" : $"{added.Count} new: {string.Join("; ", added)}";

        // 3. The woven IL: each Track call's expression (the ldstr the weave passes) and each woven try region.
        var tracked = new List<(string Where, string Call, int Line, string Text)>();
        var regions = new StringBuilder();
        using (var module = Read(path))
        {
            foreach (var type in module.GetTypes().Where(t => IsUnder(t, shape.TypeName)))
            {
                foreach (var method in type.Methods.Where(mt => mt.HasBody))
                {
                    foreach (var i in method.Body.Instructions.Where(IsTrackCall))
                    {
                        var call = (MethodReference)i.Operand;
                        var at = i;
                        for (var k = 0; k < TrackArguments[call.Name] && at != null; k++)
                            at = at.Previous;
                        var text = at?.OpCode == OpCodes.Ldstr ? (string)at.Operand : "?";
                        tracked.Add(($"{type.Name}::{method.Name}", call.Name, IntOf(i.Previous), text));
                    }

                    var points = Points(method);
                    foreach (var h in method.Body.ExceptionHandlers.Where(h => h.HandlerType == ExceptionHandlerType.Catch && Between(h.HandlerStart, h.HandlerEnd).Any(IsTrackCall)))
                    {
                        regions.AppendLine($"  {type.Name}::{method.Name} woven try IL_{h.TryStart.Offset:x4}, catch IL_{h.HandlerStart.Offset:x4}, end IL_{h.HandlerEnd?.Offset:x4}:");
                        foreach (var i in Between(h.TryStart, h.HandlerEnd))
                            regions.AppendLine("  " + Line(i, points, null, lines));
                        if (h.HandlerEnd is { } end)
                            regions.AppendLine("  " + Line(end, points, null, lines) + "   <- after the catch");
                    }
                }
            }
        }

        results.AppendLine($"weave: WeavedCount={weave.WeavedCount} MethodCount={weave.MethodCount}{(weave.SkipReason is { } skip ? " SkipReason=" + skip : "")}");
        foreach (var u in weave.Unwoven)
            results.AppendLine($"  Unwoven: {u}");
        foreach (var d in weave.DiagMessages)
            results.AppendLine($"  Diag: {d}");
        results.AppendLine(added.Count == 0 ? "  ILVerify: no error the weave added" : "  ILVerify errors the weave added:\n    " + string.Join("\n    ", added));
        results.AppendLine("assertion entry points (IsAssertionEntryPoint) in the original IL, and the sequence point each falls under:");
        results.Append(entryPoints.Length > 0 ? entryPoints.ToString() : "  none\n");
        results.AppendLine("statements, as the weaver's FindAssertionStatements and PlanStack read the original module:");
        results.Append(plan.Length > 0 ? plan.ToString() : "  none\n");
        results.AppendLine("every await in the original IL (merge point = the IsCompleted brtrue's target; stacks from StackAnalysis after SimplifyMacros):");
        results.Append(awaits.Length > 0 ? awaits.ToString() : "  none\n");
        results.AppendLine("Track calls in the woven IL (the expression is the ldstr the weave passes):");
        foreach (var group in tracked.GroupBy(t => (t.Where, t.Line, t.Text)))
        {
            var formatted = AssertionExpressionFormatter.Format(group.Key.Text);
            results.AppendLine($"  {group.Key.Where} line {group.Key.Line}: {string.Join(" + ", group.Select(t => t.Call))}");
            results.AppendLine($"    expression: \"{Esc(group.Key.Text)}\"");
            results.AppendLine($"    Format(expression) = \"{Esc(formatted)}\"");
            m.Labels[$"{group.Key.Where} L{group.Key.Line}"] = $"`{Esc(group.Key.Text)}` → \"{Esc(formatted)}\"";
        }
        if (tracked.Count == 0)
            results.AppendLine("  none");
        results.AppendLine("woven try regions (every catch whose handler calls Track):");
        results.Append(regions.Length > 0 ? regions.ToString() : "  none\n");

        // 4. The runs: the woven build, then the copy kept before the weave.
        foreach (var entry in entries)
        {
            var (thrown, notes) = RunIsolated(path, shape.TypeName, entry);
            var plainThrown = RunIsolated(plainPath, shape.TypeName, entry).Thrown;
            results.AppendLine($"run {entry}, woven: threw {Describe(thrown)}");
            foreach (var n in notes)
            {
                var (label, message) = Whole(n);
                results.AppendLine($"  note {(n.Passed ? "✓" : "✗")} \"{label}\", source {n.Source}{(message is { } mm ? ", failure message: " + mm.ReplaceLineEndings("\\n") : "")}");
            }
            if (notes.Length == 0)
                results.AppendLine("  no note drawn");
            results.AppendLine($"run {entry}, unwoven build: threw {Describe(plainThrown)}");
            m.Notes[entry] = notes.Length == 0
                ? "(none)"
                : string.Join("<br>", notes.Select(n =>
                {
                    var (label, message) = Whole(n);
                    return $"{(n.Passed ? "✓" : "✗")} {label} ({LineOf(n.Source)}){(message is { } mm ? ": " + Cut(mm.ReplaceLineEndings(" "), 80) : "")}";
                }));
            m.Threw[entry] = $"{Brief(thrown)} / {Brief(plainThrown)}";
        }
        return m;
    }

    private static string Row(Measured m, string entry) =>
        $"{m.Woven} | {string.Join("; ", m.Unwoven).Replace("|", "/")} | {Cut(m.Verify, 80).Replace("|", "/")} | {m.Notes[entry].Replace("|", "/")} | {m.Threw[entry]}";

    private static void Compare(StringBuilder s, List<(string Category, string Text)> differences, string id, string category,
        SortedDictionary<string, string> plain, SortedDictionary<string, string> dp, string? what = null)
    {
        foreach (var key in plain.Keys.Union(dp.Keys).OrderBy(k => k, StringComparer.Ordinal))
        {
            var p = plain.TryGetValue(key, out var pv) ? pv : "(none)";
            var d = dp.TryGetValue(key, out var dv) ? dv : "(none)";
            var same = Norm(p) == Norm(d);
            s.AppendLine(what == null
                ? $"| {id} | {key.Replace("|", "/")} | {p.Replace("|", "/")} | {(same ? "as plain" : d.Replace("|", "/"))} | {(same ? "same" : "**DIFFERS**")} |"
                : $"| {id} | {key.Replace("|", "/")} | {what} | {p.Replace("|", "/")} | {(same ? "as plain" : d.Replace("|", "/"))} | {(same ? "same" : "**DIFFERS**")} |");
            if (!same)
                differences.Add((category, $"{id} {key}{(what == null ? "" : " (" + what + ")")}: plain: {p}; debug-plus: {d}"));
        }
    }

    /// <summary>A value with the IL offsets and local numbers left out, which differ between builds of the same source.</summary>
    private static string Norm(string text) =>
        Regex.Replace(Regex.Replace(text, "IL_[0-9a-f]{4}", "IL_x"), @"\bV_\d+\b", "V_n");

    private static bool IsCscCommandLine(string line) =>
        (line.Contains("csc.dll", StringComparison.OrdinalIgnoreCase) || line.Contains("csc.exe", StringComparison.OrdinalIgnoreCase)) &&
        line.Contains("/out:", StringComparison.OrdinalIgnoreCase);

    /// <summary>The /debug and /optimize switches of the csc command line in a build's output.</summary>
    private static string Switches(string output)
    {
        var csc = output.ReplaceLineEndings("\n").Split('\n').Where(IsCscCommandLine).ToList();
        if (csc.Count == 0)
            return "NO CSC COMMAND LINE IN THE OUTPUT (compile skipped?)";
        return string.Join(" | ", csc.Select(l => string.Join(" ", Regex.Matches(l, @"/(debug|optimize)[^ ""]*").Select(x => x.Value))).Distinct());
    }

    /// <summary>
    /// A local copy of TestAssemblyBuilder.BuildWithSdk's approach: the fixture project and its attributes copied into a
    /// directory of the fixture's own beside a global.json pinning the SDK, restored, then built with the arguments
    /// given. Returns the assembly copied out under the fixture's name, the build's output and its command line.
    /// </summary>
    private static (string Path, string Output, string Command) BuildFixture(string name, string source, Build build, bool tunit, string verbosity)
    {
        var buildDir = Path.Combine(OutputDir, $"{name}_build");
        if (Directory.Exists(buildDir))
            Directory.Delete(buildDir, recursive: true);
        Directory.CreateDirectory(buildDir);
        var sourceFile = Path.Combine(OutputDir, $"{name}.cs");
        File.WriteAllText(sourceFile, source, Encoding.UTF8);
        File.WriteAllText(Path.Combine(buildDir, "global.json"), $$"""
            {
              "sdk": {
                "version": "{{SdkVersion}}",
                "rollForward": "latestFeature",
                "allowPrerelease": true
              }
            }
            """);
        File.WriteAllText(Path.Combine(buildDir, "Directory.Build.props"),
            "<Project><PropertyGroup><LangVersion>preview</LangVersion></PropertyGroup></Project>");
        File.WriteAllText(Path.Combine(buildDir, "Directory.Build.targets"), "<Project />");
        var csproj = Path.Combine(buildDir, "WeaverFixture.csproj");
        var project = File.ReadAllText(Path.Combine(FixturesDir, "WeaverFixture.csproj"));
        if (tunit)
        {
            const string anchor = "<PackageReference Include=\"Shouldly\" Version=\"4.3.0\" />";
            if (!project.Contains(anchor, StringComparison.Ordinal))
                throw new InvalidOperationException("WeaverFixture.csproj no longer references Shouldly 4.3.0 as the probe expects");
            project = project.Replace(anchor, anchor + "\n    <PackageReference Include=\"TUnit.Assertions\" Version=\"1.73.5\" ExcludeAssets=\"build;buildTransitive;analyzers\" />");
        }
        File.WriteAllText(csproj, project);
        File.Copy(Path.Combine(FixturesDir, "Attributes.cs"), Path.Combine(buildDir, "Attributes.cs"), overwrite: true);

        var (restoreExit, restoreOut) = Dotnet($"restore \"{csproj}\" -p:TargetFramework={Tfm}", buildDir);
        if (restoreExit != 0)
            throw new InvalidOperationException($"dotnet restore failed for '{name}':\n{restoreOut}");
        var command = $"build \"{csproj}\" {build.Arguments} -p:FixtureSource=\"{sourceFile}\" -p:TargetFramework={Tfm} -p:OutputPath=\"{buildDir}\" --no-restore -v:{verbosity}";
        var (exit, output) = Dotnet(command, buildDir);
        if (exit != 0)
            throw new InvalidOperationException($"dotnet build failed for '{name}' ({build.Label}):\n{Cut(output, 4000)}");

        var dll = Path.Combine(buildDir, "WeaverFixture.dll");
        if (!File.Exists(dll))
            throw new FileNotFoundException($"Build output not found at {dll}");
        var target = Path.Combine(OutputDir, $"{name}.dll");
        File.Copy(dll, target, overwrite: true);
        File.Copy(Path.ChangeExtension(dll, ".pdb"), Path.ChangeExtension(target, ".pdb"), overwrite: true);
        return (target, output, command);
    }

    private static (int Exit, string Output) Dotnet(string arguments, string workingDirectory)
    {
        var psi = new ProcessStartInfo("dotnet", arguments)
        {
            WorkingDirectory = workingDirectory,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true,
        };
        // As BuildWithSdk does: the test host sets these for the outer SDK, which breaks an inner build.
        foreach (var envVar in new[] { "MSBuildSDKsPath", "MSBUILD_EXE_PATH", "MSBuildExtensionsPath" })
            psi.Environment.Remove(envVar);
        using var process = Process.Start(psi)!;
        var stderr = process.StandardError.ReadToEndAsync();
        var stdout = process.StandardOutput.ReadToEnd();
        process.WaitForExit();
        return (process.ExitCode, stdout + stderr.Result);
    }

    private static string SdkRoslyn()
    {
        var compiler = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ProgramFiles), "dotnet", "sdk", SdkVersion,
            "Roslyn", "bincore", "Microsoft.CodeAnalysis.CSharp.dll");
        return File.Exists(compiler) ? FileVersionInfo.GetVersionInfo(compiler).ProductVersion ?? "?" : "? (not found at " + compiler + ")";
    }

    /// <summary>
    /// Each call the weave counts as an assertion entry point, the sequence point it falls under, and, for a visible
    /// one, the path FindAssertionStatements takes with that statement (AssertionWeaver.cs L691-722), read by calling the
    /// weave's own IsGetAwaiterOnAssertionResult and FindGetResultInstruction on the statement's instructions as it groups
    /// them: from its sequence point to the next one, hidden or not.
    /// </summary>
    private static void EntryPoints(MethodDefinition method, StringBuilder sb, string[] lines, Measured m)
    {
        var points = method.DebugInformation.SequencePoints.OrderBy(p => p.Offset).ToList();
        var offsets = points.Select(p => p.Offset).ToHashSet();
        var read = new HashSet<int>();
        foreach (var i in method.Body.Instructions)
        {
            if ((i.OpCode != OpCodes.Call && i.OpCode != OpCodes.Callvirt) || i.Operand is not MethodReference mr)
                continue;
            if (!(bool)EntryPointMethod.Invoke(null, [mr])!)
                continue;
            var sp = points.LastOrDefault(p => p.Offset <= i.Offset);
            var where = sp == null ? "no sequence point" : sp.IsHidden ? "a HIDDEN sequence point" : $"the visible sequence point L{sp.StartLine}:{sp.StartColumn}-L{sp.EndLine}:{sp.EndColumn}";
            sb.AppendLine($"  {method.DeclaringType.Name}::{method.Name} IL_{i.Offset:x4} {i.OpCode} {Operand(i)} is under {where}");
            if (sp == null)
                continue;
            if (sp.IsHidden)
            {
                var lastVisible = points.LastOrDefault(p => !p.IsHidden && p.Offset <= i.Offset);
                if (lastVisible != null)
                    m.Paths[$"{method.DeclaringType.Name}::{method.Name} L{lastVisible.StartLine} (hidden part)"] = "the call sits under a hidden sequence point after an await: in no statement";
                continue;
            }
            if (!read.Add(sp.Offset))
                continue;
            var group = new List<Instruction>();
            for (var g = method.Body.Instructions.First(x => x.Offset == sp.Offset); g != null; g = g.Next)
            {
                if (group.Count > 0 && offsets.Contains(g.Offset))
                    break;
                group.Add(g);
            }
            string path;
            if (!group.Any(x => IsCall(x, "GetAwaiter")))
                path = "synchronous statement";
            else if (!(bool)OnAssertionResultMethod.Invoke(null, [group])!)
                path = "ARGUMENT-AWAIT SKIP (L717-721): IsGetAwaiterOnAssertionResult is false, so FindAssertionStatements drops the statement: no wrap, no Unwoven entry";
            else if (FindGetResultMethod.Invoke(null, [group]) == null)
                path = "awaited, but FindGetResultInstruction finds no GetResult: wrapped as a synchronous statement";
            else
                path = "awaited assertion (IsGetAwaiterOnAssertionResult true, GetResult found)";
            m.Paths[$"{method.DeclaringType.Name}::{method.Name} L{sp.StartLine}"] = path;
            sb.AppendLine($"    the weave's path for the statement at L{sp.StartLine}: {path}");
        }
    }

    /// <summary>Every await in the method: its merge point, the stack there and before the awaiter's load, and what
    /// takes GetResult's value.</summary>
    private static void Awaits(MethodDefinition method, StackAnalysis analysis, string[] lines, StringBuilder sb, Measured m)
    {
        var visible = method.DebugInformation.SequencePoints.Where(p => !p.IsHidden).OrderBy(p => p.Offset).ToList();
        var points = Points(method);
        var k = 0;
        foreach (var br in method.Body.Instructions.Where(i => (i.OpCode == OpCodes.Brtrue || i.OpCode == OpCodes.Brtrue_S) && i.Previous is { } p && IsCall(p, "get_IsCompleted")).ToList())
        {
            k++;
            var merge = (Instruction)br.Operand;
            Instruction? getAwaiter = null;
            var n = 0;
            for (var i = br.Previous; i != null && n < 16; i = i.Previous, n++)
            {
                if (IsCall(i, "GetAwaiter"))
                {
                    getAwaiter = i;
                    break;
                }
            }
            Instruction? getResult = null;
            n = 0;
            for (var i = merge; i != null && n < 16; i = i.Next, n++)
            {
                if (IsCall(i, "GetResult"))
                {
                    getResult = i;
                    break;
                }
            }
            var load = getResult?.Previous;
            var sp = visible.LastOrDefault(p => p.Offset <= (getAwaiter ?? br).Offset);
            var line = sp == null ? "L?" : $"L{sp.StartLine}";
            var mergeStack = analysis.StackBefore(merge);
            var loadStack = load == null ? null : analysis.StackBefore(load);
            var after = getResult?.Next;
            var afterStartsStatement = after != null && points.TryGetValue(after.Offset, out var ap) && !ap.IsHidden;
            var key = $"{method.DeclaringType.Name}::{method.Name} await#{k} {line}";
            var mergeText = $"{merge.OpCode} {Operand(merge)}".TrimEnd();
            m.AwaitMerge[key] = mergeText;
            m.AwaitMergeStack[key] = StackText(mergeStack);
            m.MergeShape[key] = $"merge {mergeText}, stack {StackText(mergeStack)}; awaiter load {(load == null ? "?" : $"{load.OpCode} {Operand(load)}")}, stack before it {StackText(loadStack)}";
            m.ResultTaker[key] = $"{(after == null ? "?" : $"{after.OpCode} {Operand(after)}".TrimEnd())}{(afterStartsStatement ? " (starts the next statement: the value crosses into it on the stack)" : "")}";
            m.Awaits[key] = $"merge {mergeText}, stack {StackText(mergeStack)}; awaiter load {(load == null ? "?" : $"{load.OpCode} {Operand(load)}")}, stack before it {StackText(loadStack)}; " +
                            $"then {(after == null ? "?" : $"{after.OpCode} {Operand(after)}".TrimEnd())}{(afterStartsStatement ? " (the next statement's first instruction: the value crosses into it on the stack)" : "")}";
            sb.AppendLine($"  {key} \"{(sp == null ? "" : SourceLine(lines, sp.StartLine))}\": GetAwaiter {(getAwaiter == null ? "?" : $"IL_{getAwaiter.Offset:x4} {Operand(getAwaiter)}")}; IsCompleted brtrue IL_{br.Offset:x4} -> merge point IL_{merge.Offset:x4} {mergeText}");
            sb.AppendLine($"    stack at the merge point: {StackText(mergeStack)}");
            sb.AppendLine($"    awaiter load before GetResult: {(load == null ? "?" : $"IL_{load.Offset:x4} {load.OpCode} {Operand(load)}")}; stack before it: {StackText(loadStack)}");
            sb.AppendLine($"    GetResult {(getResult == null ? "?" : $"IL_{getResult.Offset:x4} {Operand(getResult)}")}; then {(after == null ? "?" : $"IL_{after.Offset:x4} {after.OpCode} {Operand(after)}")}{(afterStartsStatement ? ", which starts the next visible sequence point" : "")}");
        }
    }

    private static string StackText(StackAnalysis.Slot[]? stack) => stack == null
        ? "unreached"
        : stack.Length == 0
            ? "empty"
            : $"[{stack.Length}] " + string.Join(", ", stack.Select(s => $"{Short(s.Type)} (pushed by {(s.Producer is { } p ? $"IL_{p.Offset:x4} {p.OpCode} {Operand(p)}".TrimEnd() : "joined paths")})"));

    /// <summary><see cref="WovenRun.Run"/> in a collectible load context of its own: every build is named WeaverFixture,
    /// which the default context loads once.</summary>
    private static (Exception? Thrown, WovenRun.Note[] Notes) RunIsolated(string path, string typeName, string methodName)
    {
        var context = new System.Runtime.Loader.AssemblyLoadContext($"{Path.GetFileNameWithoutExtension(path)}_{Guid.NewGuid():N}", isCollectible: true);
        try
        {
            var type = context.LoadFromAssemblyPath(path).GetType(typeName)!;
            var method = type.GetMethod(methodName)!;
            var instance = method.IsStatic ? null : Activator.CreateInstance(type);
            var testId = $"{methodName}_{Guid.NewGuid():N}";
            Exception? thrown;
            using (TestIdentityScope.Begin(testId, testId))
            {
                thrown = Record.Exception(() =>
                {
                    if (method.Invoke(instance, null) is Task task)
                        task.GetAwaiter().GetResult();
                });
            }
            if (thrown is TargetInvocationException { InnerException: { } inner })
                thrown = inner;
            return (thrown, WovenRun.NotesFor(testId));
        }
        finally
        {
            context.Unload();
        }
    }

    /// <summary>PlanStack's verdict on the statement, with its entry and exit stacks, written to the results; returns
    /// the same for the comparison.</summary>
    private static (string Display, string Verdict, string Stacks, string? Next) PlanOf(MethodDefinition method, AssertionStatement st, StackAnalysis analysis, StringBuilder sb)
    {
        var args = new object?[] { method, st, analysis, null, null };
        var reason = (string?)PlanStackMethod.Invoke(null, args);
        var entry = (StackAnalysis.Slot[])args[3]!;
        var exit = (StackAnalysis.Slot[])args[4]!;
        var decision = reason == null ? "woven" : "REFUSED: " + reason;
        var stacks = $"entry=[{Slots(entry)}] exit=[{Slots(exit)}]";
        var verdict = $"PlanStack {decision}; {stacks}";
        sb.AppendLine("    " + verdict);
        if (!st.IsAwaited || st.GetResultInstruction == null)
            return ("sync; " + verdict, "sync, " + decision, stacks, null);
        var (start, end) = ((Instruction, Instruction))AwaitedRangeMethod.Invoke(null, [st.GetResultInstruction])!;
        sb.AppendLine($"    AwaitedRange: IL_{start.Offset:x4} {start.OpCode} {Operand(start)} .. IL_{end.Offset:x4} {end.OpCode} {Operand(end)}; then IL_{end.Next?.Offset:x4} {end.Next?.OpCode} {Operand(end.Next)}");
        sb.AppendLine("    stack before the range's start: " + StackText(analysis.StackBefore(start)));
        var next = $"{start.OpCode} .. {end.OpCode}, then {end.Next?.OpCode} {Operand(end.Next)}".TrimEnd();
        return ($"awaited; {verdict}; AwaitedRange {next}", "awaited, " + decision, stacks, next);
    }

    private static string Cut(string text, int max) => text.Length <= max ? text : text[..max] + "...";

    private static string Esc(string text) => text.Replace("\r", "\\r").Replace("\n", "\\n");

    private static string LineOf(string? source) =>
        source == null ? "no source" : source.LastIndexOf(":L", StringComparison.Ordinal) is var at and >= 0 ? source[(at + 1)..] : source;

    private const string JoinMarker = "<U+200B>";

    /// <summary>A note's label and failure message with the note's wrapped lines rejoined, as
    /// Kronikol.PlantUml.DiagramWidth.RejoinMarkedLines does (two join markers for a break at a space, one for a cut
    /// inside a word); the label is the first line after that.</summary>
    private static (string Label, string? Message) Whole(WovenRun.Note note)
    {
        var lines = (note.Label + (note.Message is { } m ? "\n" + m : "")).Split('\n');
        var text = new StringBuilder();
        for (var i = 0; i < lines.Length; i++)
        {
            var line = lines[i];
            var last = i == lines.Length - 1;
            if (!last && line.EndsWith(JoinMarker + JoinMarker, StringComparison.Ordinal))
                text.Append(line, 0, line.Length - 2 * JoinMarker.Length).Append(' ');
            else if (!last && line.EndsWith(JoinMarker, StringComparison.Ordinal))
                text.Append(line, 0, line.Length - JoinMarker.Length);
            else
                text.Append(line).Append(last ? "" : "\n");
        }
        var whole = text.ToString();
        var at = whole.IndexOf('\n');
        return at < 0 ? (whole, null) : (whole[..at], whole[(at + 1)..]);
    }

    private static string SourceLine(string[] lines, int line) =>
        line >= 1 && line <= lines.Length ? lines[line - 1].Trim() : "";

    /// <summary>How the awaited wrap loads a captured value at the merge point.</summary>
    private static string Describe(CapturedVariable cv, MethodDefinition method)
    {
        if (cv.ClosureField != null && cv.ClosureLocalIndex >= 0)
            return $"'{cv.Name}': ldloc V_{cv.ClosureLocalIndex} (a local of {method.DeclaringType.Name}::{method.Name}, {Short(method.Body.Variables[cv.ClosureLocalIndex].VariableType)}); ldfld {cv.ClosureField.Name}";
        if (cv.ClosureField != null && cv.StateField != null)
            return $"'{cv.Name}': ldarg.0; ldfld {cv.StateField.Name} ({Short(cv.StateField.FieldType)}); ldfld {cv.ClosureField.Name}";
        if (cv.StateField != null)
            return $"'{cv.Name}': ldarg.0; ldfld {cv.StateField.Name}";
        if (cv.ParameterIndex >= 0)
            return $"'{cv.Name}': ldarg {cv.ParameterIndex}";
        return $"'{cv.Name}': ldloc V_{cv.LocalIndex}";
    }

    /// <summary>The original IL from the statement's last GetAwaiter to the next visible sequence point after GetResult.</summary>
    private static string? Excerpt(MethodDefinition method, AssertionStatement st, string[] lines)
    {
        var points = Points(method);
        Instruction? getAwaiter = null;
        for (var i = st.FirstInstruction; i != null; i = i.Next)
        {
            if (i != st.FirstInstruction && points.TryGetValue(i.Offset, out var p) && !p.IsHidden)
                break;
            if (IsCall(i, "GetAwaiter"))
                getAwaiter = i;
        }
        if (getAwaiter == null)
            return null;

        var analysis = SafeAnalyze(method);
        var sb = new StringBuilder();
        var seenGetResult = false;
        var count = 0;
        for (var i = getAwaiter; i != null && count++ < 160; i = i.Next)
        {
            var visible = points.TryGetValue(i.Offset, out var p) && !p.IsHidden;
            if (visible && i != getAwaiter && seenGetResult)
            {
                sb.AppendLine("    " + Line(i, points, analysis, lines) + "   <- next visible sequence point");
                break;
            }
            sb.AppendLine("    " + Line(i, points, analysis, lines));
            if (IsCall(i, "GetResult"))
                seenGetResult = true;
        }
        return sb.ToString();
    }

    private static string Line(Instruction i, Dictionary<int, SequencePoint> points, StackAnalysis? analysis, string[] lines)
    {
        points.TryGetValue(i.Offset, out var p);
        var mark = p == null ? "" : p.IsHidden ? "SP hidden" : $"SP L{p.StartLine}:{p.StartColumn}-L{p.EndLine}:{p.EndColumn}";
        var stack = analysis == null
            ? ""
            : analysis.StackBefore(i) is { } s
                ? $"  stack[{s.Length}]{(s.Length > 0 ? " " + string.Join(", ", s.Select(x => Short(x.Type))) : "")}"
                : "  stack[unreached]";
        var text = $"{mark,-24} IL_{i.Offset:x4}: {i.OpCode.Name,-14} {Operand(i),-72}{stack}";
        if (p is { IsHidden: false } && p.StartLine >= 1 && p.StartLine <= lines.Length)
            text += $"   // {lines[p.StartLine - 1].Trim()}";
        return text;
    }

    private static string Operand(Instruction? i) => i?.Operand switch
    {
        null => "",
        Instruction t => $"IL_{t.Offset:x4}",
        Instruction[] ts => string.Join(",", ts.Select(t => $"IL_{t.Offset:x4}")),
        GenericInstanceMethod gm => $"{Short(gm.DeclaringType)}::{gm.Name}<{string.Join(",", gm.GenericArguments.Select(Short))}>",
        MethodReference mr => $"{Short(mr.DeclaringType)}::{mr.Name}",
        FieldReference fr => $"{Short(fr.DeclaringType)}::{fr.Name}",
        VariableDefinition v => $"V_{v.Index}",
        ParameterDefinition pd => pd.Index < 0 || string.IsNullOrEmpty(pd.Name) ? "this" : pd.Name,
        string s => "\"" + s.ReplaceLineEndings("\\n") + "\"",
        TypeReference t => Short(t),
        _ => i.Operand.ToString() ?? "",
    };

    private static string Short(TypeReference? t) => t == null
        ? "?"
        : t.FullName.Replace("System.Threading.Tasks.", "").Replace("System.Runtime.CompilerServices.", "").Replace("System.", "");

    private static string Slots(StackAnalysis.Slot[] slots) => string.Join(", ", slots.Select(s => Short(s.Type)));

    private static Dictionary<int, SequencePoint> Points(MethodDefinition method) =>
        method.DebugInformation.SequencePoints.GroupBy(p => p.Offset).ToDictionary(g => g.Key, g => g.First());

    private static StackAnalysis? SafeAnalyze(MethodDefinition method)
    {
        try
        {
            return StackAnalysis.Analyze(method.Body);
        }
        catch
        {
            return null;
        }
    }

    private static IEnumerable<Instruction> Between(Instruction start, Instruction? end)
    {
        for (var i = start; i != null && i != end; i = i.Next)
            yield return i;
    }

    private static bool IsUnder(TypeDefinition type, string root) =>
        type.FullName == root || type.FullName.StartsWith(root + "/", StringComparison.Ordinal);

    private static bool IsCall(Instruction i, string name) =>
        (i.OpCode == OpCodes.Call || i.OpCode == OpCodes.Callvirt) && i.Operand is MethodReference mr && mr.Name == name;

    private static bool IsTrackCall(Instruction i) =>
        i.OpCode == OpCodes.Call && i.Operand is MethodReference mr &&
        mr.DeclaringType.FullName == "Kronikol.Tracking.Track" && TrackArguments.ContainsKey(mr.Name);

    private static int IntOf(Instruction? i) => i == null ? -1 : i.OpCode.Code switch
    {
        Code.Ldc_I4 => (int)i.Operand,
        Code.Ldc_I4_S => (sbyte)i.Operand,
        Code.Ldc_I4_0 => 0,
        Code.Ldc_I4_1 => 1,
        Code.Ldc_I4_2 => 2,
        Code.Ldc_I4_3 => 3,
        Code.Ldc_I4_4 => 4,
        Code.Ldc_I4_5 => 5,
        Code.Ldc_I4_6 => 6,
        Code.Ldc_I4_7 => 7,
        Code.Ldc_I4_8 => 8,
        _ => -1,
    };

    private static string Describe(Exception? ex)
    {
        if (ex == null)
            return "nothing";
        var frames = (ex.StackTrace ?? "").ReplaceLineEndings("\n").Split('\n')
            .Where(l => l.Contains(" in ", StringComparison.Ordinal) || l.Contains("Probe", StringComparison.Ordinal) || l.Contains("---", StringComparison.Ordinal))
            .Take(8)
            .Select(l => "\n      " + l.Trim());
        return $"{ex.GetType().FullName}: {ex.Message.ReplaceLineEndings("\\n")}{string.Concat(frames)}";
    }

    private static string Brief(Exception? ex) => ex == null ? "no" : ex.GetType().Name;

    private static string ShortReason(string unwoven)
    {
        var at = unwoven.IndexOf(", line ", StringComparison.Ordinal);
        return (at < 0 ? unwoven : unwoven[(at + 2)..]).Replace("|", "/");
    }

    private static string Numbered(string source)
    {
        var all = source.ReplaceLineEndings("\n").Split('\n');
        return string.Join("\n", all.Select((l, n) => $"{n + 1,3}: {l}")) + "\n";
    }

    private static string Indent(string body) =>
        string.Join("\n", body.ReplaceLineEndings("\n").Split('\n').Select(l => l.Length == 0 ? l : "    " + l));

        Directory.CreateDirectory(root);
        return root;
    }

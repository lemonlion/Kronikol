  1: #nullable enable
  2: using System;
  3: using System.Threading.Tasks;
  4: using Kronikol.Tracking;
  5: using Shouldly;

  7: [assembly: TrackAssertions]
  8: 
  9: public sealed class OrderRequest
 10: {
 11:     public string Id { get; init; } = "";
 12: }
 13: 
 14: public sealed class OrdersClient
 15: {
 16:     public async Task<string> GetOrderAsync(OrderRequest request)
 17:     {
 18:         await Task.Yield();
 19:         return request.Id == "known" ? "order" : throw new InvalidOperationException("not found");
 20:     }
 21: 
 22:     public string GetOrder(OrderRequest request) =>
 23:         request.Id == "known" ? "order" : throw new InvalidOperationException("not found");
 24: }
 25: 
 26: public class Probe
 27: {
 28:     private readonly OrdersClient _client = new();
 29: 
 30:     public async Task AssertionPasses()
 31:     {
 32:         await Ok(new OrderRequest { Id = "known" });
 33:     }
 34: 
 35:     public async Task AssertionFails()
 36:     {
 37:         await Ok(new OrderRequest { Id = "missing" });
 38:     }
 39: 
 40:     private Task Ok(OrderRequest r) => Should.NotThrowAsync(() => _client.GetOrderAsync(r));
 41: }

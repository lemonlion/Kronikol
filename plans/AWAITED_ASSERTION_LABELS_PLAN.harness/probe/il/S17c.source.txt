 10: public sealed class OrderRequest

 16: {

 18: 
 19:     public async Task<string> GetNameAsync()
 20:     {
 21:         await Task.Yield();
 22:         return "order";
 23:     }
 24:

 26:     {
 27:         await Task.Yield();

 29:     }
 30:

 32:     {
 33:         await Task.Yield();
 34:         return ExpectedMessage;
 35:     }
 36: }
